### 1.4 Ranking de Vendedores/Equipe

```sql
-- GERENTE (agrupa por vendedor com sub-consultas de atendimentos e pedidos)
SELECT U.CODUSUR, MAX(U.NOME) AS NOME,
    (SELECT COUNT(DISTINCT TELEFONE_CLIENTE) FROM CANAL_MENSAGENS
     WHERE CODUSUR = U.CODUSUR AND {filtro_data}) AS ATENDIMENTOS,
    (SELECT COUNT(DISTINCT NUMPED) FROM PCPEDC
     WHERE CODUSUR = U.CODUSUR AND {filtro_data}) AS PEDIDOS,
    (SELECT NVL(SUM(VLTOTAL), 0) FROM PCPEDC
     WHERE CODUSUR = U.CODUSUR AND {filtro_data}) AS VLTOTAL,
    MAX(S.CODSUPERVISOR) AS CODSUPERVISOR,
    MAX(S.NOME) AS NOME_SUPERVISOR
FROM PCUSUARI U
JOIN PCSUPERV S ON U.CODSUPERVISOR = S.CODSUPERVISOR
WHERE S.CODGERENTE = (SELECT CODGERENTE FROM PCGERENTE WHERE COD_CADRCA = :cod)
GROUP BY U.CODUSUR;

-- VENDEDOR (ranking por cliente)
SELECT C.CODCLI AS CODUSUR, MAX(CLI.CLIENTE) AS NOME,
    (SELECT COUNT(DISTINCT TELEFONE_CLIENTE) FROM CANAL_MENSAGENS M
     JOIN VW_CANAL_CLIENTES V ON M.TELEFONE_CLIENTE = V.TELEFONE
     WHERE V.CODCLI = C.CODCLI AND M.CODUSUR = :cod AND {filtro_data}) AS ATENDIMENTOS,
    COUNT(DISTINCT C.NUMPED) AS PEDIDOS,
    NVL(SUM(C.VLTOTAL), 0) AS VLTOTAL
FROM PCPEDC C
JOIN PCCLIENT CLI ON C.CODCLI = CLI.CODCLI
WHERE C.CODUSUR = :cod AND {filtro_data}
GROUP BY C.CODCLI;
```

### 3.3 Histórico de pedidos / produtos comprados

```sql
-- PR    = PCPRODUT    : descrição e unidade (sem preço vigente)
-- TAB   = PCTABCLI + PCEMBALAGEM : preço da tabela específica do cliente (se houver)
-- TABPR = PCTABPR     : preço vigente padrão por produto/região (fallback)
-- PROMO = PCPRECOPROM : preço promocional vigente hoje (prioridade máxima)
SELECT
    M.CODPROD,
    PR.DESCRICAO,
    PR.UNIDADE,
    SUM(M.QT)                                            AS QT_TOTAL,
    MAX(M.DTMOV)                                         AS ULTIMA_COMPRA,
    MAX(M.PUNIT)                                         AS PRECO_ULTIMA_COMPRA,
    -- Prioridade: promoção vigente > tabela do cliente > preço padrão (PCTABPR)
    MAX(NVL(PROMO.PRECOFIXO, NVL(TAB.PVENDA, TABPR.PVENDA))) AS PRECO_ATUAL,
    MAX(CASE WHEN PROMO.CODPROD IS NOT NULL THEN 1 ELSE 0 END) AS TEM_PROMO,
    -- Flag de embalagem FV ativa
    MAX(CASE
        WHEN NVL(EMB.ENVIAFV, 'N') = 'S'
         AND EMB.DTINATIVO IS NULL THEN 1
        ELSE 0
    END)                                                 AS TEM_FV
FROM PCMOV M
-- PR: dados cadastrais do produto (descrição e unidade)
JOIN PCPRODUT PR
    ON PR.CODPROD = M.CODPROD
-- TABPR: preço vigente padrão — chave: CODPROD + NUMREGIAO (via config NUMREGIAO)
LEFT JOIN PCTABPR TABPR
    ON  TABPR.CODPROD  = M.CODPROD
    AND TABPR.NUMREGIAO = (SELECT VALOR FROM CANAL_CONFIGURACOES WHERE CHAVE = 'NUMREGIAO')
-- TAB: tabela de preço específica do cliente (PCTABCLI → PCEMBALAGEM)
LEFT JOIN (
    SELECT TC.CODCLI, TC.NUMTABELA
    FROM PCTABCLI TC
    WHERE TC.CODCLI = :codcli
      AND ROWNUM = 1
) TABCLI ON TABCLI.CODCLI = M.CODCLI
LEFT JOIN PCEMBALAGEM TAB
    ON  TAB.CODPROD   = M.CODPROD
    AND TAB.EMBALAGEM = TABCLI.NUMTABELA
-- PROMO: promoção de preço vigente hoje (por CODPROD ou CODAUXILIAR)
LEFT JOIN PCPRECOPROM PROMO
    ON  (PROMO.CODPROD = M.CODPROD OR PROMO.CODAUXILIAR = PR.CODAUXILIAR)
    AND TRUNC(SYSDATE) BETWEEN TRUNC(PROMO.DTINICIOVIGENCIA)
                           AND TRUNC(PROMO.DTFIMVIGENCIA)
-- EMB: verifica se existe embalagem FV ativa para o produto
LEFT JOIN PCEMBALAGEM EMB
    ON  EMB.CODPROD = M.CODPROD
    AND NVL(EMB.ENVIAFV, 'N') = 'S'
    AND EMB.DTINATIVO IS NULL
WHERE M.CODCLI = :codcli
GROUP BY M.CODPROD, PR.DESCRICAO, PR.UNIDADE
ORDER BY ULTIMA_COMPRA DESC;
```

> **PROMO** = `PCPRECOPROM`: preço fixo promocional — vigente quando `TRUNC(SYSDATE) BETWEEN DTINICIOVIGENCIA AND DTFIMVIGENCIA`. **Prioridade máxima.**
> **TAB** = `PCTABCLI` + `PCEMBALAGEM`: preço da tabela específica do cliente.
> **TABPR** = `PCTABPR`: preço vigente padrão por produto/região (`NUMREGIAO = 0`). Fallback final.
> **PR** = `PCPRODUT`: apenas descrição e unidade — não armazena preço vigente.

### 5.1 KPIs do SAC (cards)

```sql
SELECT COUNT(*) FROM CANAL_SAC_TICKETS WHERE STATUS = 'ABERTO';

SELECT COUNT(*) FROM CANAL_SAC_TICKETS
WHERE STATUS IN ('FECHADO', 'FINALIZADO') AND TRUNC(ATUALIZADO_EM) = TRUNC(SYSDATE);

SELECT NVL(AVG(NOTA_AVALIACAO), 0) FROM CANAL_SAC_TICKETS
WHERE NOTA_AVALIACAO IS NOT NULL AND TRUNC(ATUALIZADO_EM, 'MM') = TRUNC(SYSDATE, 'MM');

-- SLA médio em horas
SELECT NVL(AVG((CAST(DATA_RESOLUCAO AS DATE) - CAST(CRIADO_EM AS DATE)) * 24), 0)
FROM CANAL_SAC_TICKETS
WHERE STATUS IN ('FECHADO', 'FINALIZADO', 'RESOLVIDO') AND DATA_RESOLUCAO IS NOT NULL;

-- Volume por departamento
SELECT d.NOME, COUNT(t.ID) FROM CANAL_SAC_TICKETS t
JOIN CANAL_SAC_DEPARTAMENTOS d ON t.DEPARTAMENTO_ID = d.ID GROUP BY d.NOME;

-- Top 5 clientes
SELECT NVL(c.FANTASIA, c.CLIENTE), COUNT(t.ID)
FROM CANAL_SAC_TICKETS t JOIN PCCLIENT c ON t.CODCLI = c.CODCLI
GROUP BY NVL(c.FANTASIA, c.CLIENTE)
ORDER BY COUNT(t.ID) DESC FETCH FIRST 5 ROWS ONLY;

-- Top 5 vendedores
SELECT u.NOME, COUNT(t.ID) FROM CANAL_SAC_TICKETS t
JOIN PCCLIENT c ON t.CODCLI = c.CODCLI
JOIN PCUSUARI u ON c.CODUSUR1 = u.CODUSUR
GROUP BY u.NOME ORDER BY COUNT(t.ID) DESC FETCH FIRST 5 ROWS ONLY;

-- Taxa de resolução mensal
SELECT COUNT(CASE WHEN STATUS IN ('FECHADO','FINALIZADO','RESOLVIDO') THEN 1 END) AS RESOLVIDOS,
       COUNT(*) AS TOTAL
FROM CANAL_SAC_TICKETS WHERE TRUNC(CRIADO_EM, 'MM') = TRUNC(SYSDATE, 'MM');

-- NPS mensal
SELECT COUNT(CASE WHEN NOTA_AVALIACAO >= 9 THEN 1 END) AS PROMOTORES,
       COUNT(CASE WHEN NOTA_AVALIACAO <= 6 THEN 1 END) AS DETRATORES,
       COUNT(NOTA_AVALIACAO) AS TOTAL_NOTAS
FROM CANAL_SAC_TICKETS
WHERE NOTA_AVALIACAO IS NOT NULL AND TRUNC(CRIADO_EM, 'MM') = TRUNC(SYSDATE, 'MM');

SELECT COUNT(*) FROM CANAL_SAC_TICKETS WHERE TRUNC(CRIADO_EM) = TRUNC(SYSDATE);
```

### 5.2 Listar tickets (com filtro de acesso)

```sql
WITH LATEST_LOG AS (
    SELECT TELEFONE, DOCUMENTO_INFORMADO, CODCLI_LOCALIZADO
    FROM (
        SELECT TELEFONE, DOCUMENTO_INFORMADO, CODCLI_LOCALIZADO,
               ROW_NUMBER() OVER(PARTITION BY TELEFONE ORDER BY DATA_HORA DESC) AS RN
        FROM CANAL_LOG_IDENTIFICACAO_CLIENTE
    ) WHERE RN = 1
)
SELECT t.ID, t.TELEFONE, t.CODCLI, t.DEPARTAMENTO_ID,
       CASE WHEN p.NOME IS NOT NULL THEN p.NOME || ' / ' || d.NOME ELSE d.NOME END AS DEPARTAMENTO,
       t.DESCRICAO, t.STATUS, t.CRIADO_EM, t.ATUALIZADO_EM, t.NOTA_AVALIACAO,
       NVL(c.FANTASIA, c.CLIENTE) AS NOME_CLIENTE, c.CGCENT,
       l.DOCUMENTO_INFORMADO AS LOG_CNPJ, l.CODCLI_LOCALIZADO AS LOG_CODCLI,
       t.DATA_AGENDAMENTO, t.AGENDAMENTO_CODPROD, t.AGENDAMENTO_QTDE,
       t.AGENDAMENTO_MOTORISTA_NOME, t.AGENDAMENTO_MOTORISTA_TEL, t.AGENDAMENTO_ENVIADO,
       prod.DESCRICAO AS PRODUTO_NOME, t.MENTIONED_MATRICULA, t.MENTIONED_NOME
FROM CANAL_SAC_TICKETS t
LEFT JOIN CANAL_SAC_DEPARTAMENTOS d ON t.DEPARTAMENTO_ID = d.ID
LEFT JOIN CANAL_SAC_DEPARTAMENTOS p ON d.DEPARTAMENTO_PAI_ID = p.ID
LEFT JOIN LATEST_LOG l ON t.TELEFONE = l.TELEFONE
LEFT JOIN PCCLIENT c ON c.CODCLI = NVL(t.CODCLI, l.CODCLI_LOCALIZADO)
LEFT JOIN PCPRODUT prod ON t.AGENDAMENTO_CODPROD = prod.CODPROD
WHERE 1=1
  AND (
    t.DEPARTAMENTO_ID IN (SELECT DEPARTAMENTO_ID FROM CANAL_SAC_ACESSOS WHERE MATRICULA = :matricula)
    OR d.DEPARTAMENTO_PAI_ID IN (SELECT DEPARTAMENTO_ID FROM CANAL_SAC_ACESSOS WHERE MATRICULA = :matricula)
  )
  AND t.STATUS = :st  -- opcional
ORDER BY t.CRIADO_EM DESC;
```

SELECT t.TELEFONE, t.STATUS, d.NOME, p.NOME, t.DESCRICAO, t.CODCLI

LEFT JOIN CANAL_SAC_DEPARTAMENTOS d ON t.DEPARTAMENTO_ID = d.ID
LEFT JOIN CANAL_SAC_DEPARTAMENTOS p ON d.DEPARTAMENTO_PAI_ID = p.ID
WHERE t.ID = :id;

-- Instância WhatsApp do SAC BOT
SELECT VALOR FROM CANAL_CONFIGURACOES WHERE CHAVE = 'SAC_BOT_CODUSUR';
SELECT INSTANCE_NAME, API_TOKEN,
    COALESCE(API_URL, (SELECT VALOR FROM CANAL_CONFIGURACOES WHERE CHAVE = 'EVOLUTION_API_URL'))
FROM CANAL_TOKENS_EVOLUTION WHERE CODUSUR = :cod;

-- Gravar mensagem enviada
INSERT INTO CANAL_MENSAGENS
    (ID_MENSAGEM, CODUSUR, TELEFONE_CLIENTE, SENTIDO, TEXTO, STATUS, DATA_HORA, TICKET_ID, GROK_USADO)
VALUES (:id_msg, :cod, :tel, 'OUT', :txt, :status, SYSDATE, :tId, :grok);

-- Atualizar status do ticket
UPDATE CANAL_SAC_TICKETS SET STATUS = 'EM ATENDIMENTO', ATUALIZADO_EM = SYSDATE WHERE ID = :id;

-- Manter cliente em atendimento (bot)
MERGE INTO CANAL_BOT_STATE dest
USING (SELECT :tel AS TELEFONE FROM DUAL) src
ON (dest.TELEFONE = src.TELEFONE)
WHEN MATCHED THEN UPDATE SET ESTADO_ATUAL = 'AGUARDANDO_TICKET_ACAO',
    DADOS_TEMPORARIOS = :dados, ATUALIZADO_EM = SYSDATE
WHEN NOT MATCHED THEN INSERT (TELEFONE, ESTADO_ATUAL, DADOS_TEMPORARIOS, ATUALIZADO_EM)
VALUES (:tel, 'AGUARDANDO_TICKET_ACAO', :dados, SYSDATE);

-- Notificar vendedor do cliente (se houver instância configurada)
SELECT CODUSUR1 FROM PCCLIENT WHERE CODCLI = :c AND ROWNUM = 1;
SELECT TELEFONE1 FROM PCUSUARI WHERE CODUSUR = :c AND ROWNUM = 1;
SELECT CELULAR FROM PCEMPR WHERE MATRICULA = :c AND ROWNUM = 1;
```

LEFT JOIN CANAL_TOKENS_EVOLUTION T
    ON TO_CHAR(E.MATRICULA) = T.CODUSUR AND T.CARGO = 'ATENDENTE'
WHERE E.SITUACAO = 'A' ORDER BY E.NOME;
```

### 16.3 Vincular WhatsApp (Evolution API)

```sql
MERGE INTO CANAL_TOKENS_EVOLUTION T
USING (SELECT :codusur AS CODUSUR, :cargo AS CARGO, :api_token AS API_TOKEN,
              :instance_name AS INSTANCE_NAME, :api_url AS API_URL,
              :nome_atendente AS NOME_ATENDENTE FROM DUAL) S
ON (T.CODUSUR = S.CODUSUR AND T.CARGO = S.CARGO)
WHEN MATCHED THEN UPDATE SET
    T.API_TOKEN = S.API_TOKEN, T.INSTANCE_NAME = S.INSTANCE_NAME,
    T.API_URL = S.API_URL, T.NOME_ATENDENTE = S.NOME_ATENDENTE, T.DATA_ATUALIZACAO = SYSDATE
WHEN NOT MATCHED THEN
    INSERT (CODUSUR, CARGO, API_TOKEN, INSTANCE_NAME, API_URL, NOME_ATENDENTE)
    VALUES (S.CODUSUR, S.CARGO, S.API_TOKEN, S.INSTANCE_NAME, S.API_URL, S.NOME_ATENDENTE);
```

### 17.1 Catálogo Geral e Oportunidades

> Retorna o catálogo completo, priorizando Oportunidades (promoções vigentes) no topo da lista.

```sql
SELECT 
    A.CODPROD,
    A.CODAUXILIAR,
    D.DESCRICAO,
    A.UNIDADE,
    A.EMBALAGEM,
    A.ENVIAFV,
    A.QTUNIT,
    A.TIPOEMBALAGEM,
    D.PESOLIQ,
    -- Prioriza o preço promocional; se nulo, traz a tabela normal
    NVL(B.PRECOFIXO, C.PTABELA) AS PRECO_VENDA,
    -- Identificador se o preço exibido é promocional ou regular
    CASE 
        WHEN B.PRECOFIXO IS NOT NULL THEN 'OPORTUNIDADE' 
        ELSE '' 
    END AS TIPO_PRECO,
    B.DTINICIOVIGENCIA,
    B.DTFIMVIGENCIA
FROM PCEMBALAGEM A
LEFT JOIN PCPRECOPROM B 
       ON A.CODAUXILIAR = B.CODAUXILIAR 
      AND B.NUMREGIAO = :numregiao
      AND TRUNC(SYSDATE) BETWEEN B.DTINICIOVIGENCIA AND B.DTFIMVIGENCIA
      AND B.DTINICIOVIGENCIA IS NOT NULL
      AND B.DTFIMVIGENCIA IS NOT NULL
LEFT JOIN PCTABPR C 
       ON A.CODPROD = C.CODPROD 
      AND C.NUMREGIAO = :numregiao
JOIN PCPRODUT D 
  ON A.CODPROD = D.CODPROD
JOIN PCEST E 
  ON A.CODPROD = E.CODPROD 
 AND A.CODFILIAL = E.CODFILIAL 
 AND E.CODFILIAL = :filial 
 AND (E.QTESTGER - E.QTBLOQUEADA) > 0
WHERE A.ENVIAFV = 'S'
  AND A.DTINATIVO IS NULL
ORDER BY 
    CASE WHEN B.PRECOFIXO IS NOT NULL THEN 0 ELSE 1 END,
    A.CODPROD;
```

### 17.2 Apenas Oportunidades (Promoções)

> Utiliza a mesma estrutura base, mas exibe apenas os produtos que têm uma promoção ativa. Ideal para exibir os "destaques" do catálogo.

```sql
SELECT 
    A.CODPROD,
    A.CODAUXILIAR,
    D.DESCRICAO,
    A.UNIDADE,
    A.EMBALAGEM,
    A.ENVIAFV,
    A.QTUNIT,
    A.TIPOEMBALAGEM,
    D.PESOLIQ,
    B.PRECOFIXO AS PRECO_VENDA,
    'OPORTUNIDADE' AS TIPO_PRECO,
    B.DTINICIOVIGENCIA,
    B.DTFIMVIGENCIA
FROM PCEMBALAGEM A
-- Usamos JOIN direto em vez de LEFT JOIN para garantir que só retorne oportunidades
JOIN PCPRECOPROM B 
       ON A.CODAUXILIAR = B.CODAUXILIAR 
      AND B.NUMREGIAO = :numregiao
      AND TRUNC(SYSDATE) BETWEEN B.DTINICIOVIGENCIA AND B.DTFIMVIGENCIA
      AND B.DTINICIOVIGENCIA IS NOT NULL
      AND B.DTFIMVIGENCIA IS NOT NULL
JOIN PCPRODUT D 
  ON A.CODPROD = D.CODPROD
JOIN PCEST E 
  ON A.CODPROD = E.CODPROD 
 AND A.CODFILIAL = E.CODFILIAL 
 AND E.CODFILIAL = :filial 
 AND (E.QTESTGER - E.QTBLOQUEADA) > 0
WHERE A.ENVIAFV = 'S'
  AND A.DTINATIVO IS NULL
ORDER BY A.CODPROD;
```
